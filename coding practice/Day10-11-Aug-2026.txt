## Ansible - Create Docker Containers for Multiple Application 

```yaml 
- name: Setup Docker Containers for Multiple Apps 
  hosts: all
  become: yes 
  tasks: 
    - name: Pull Web App Image 
      docker_image: 
        name: my-web-app 
        source: pull 
    
    - name: Run Web App Container 
      docker_container: 
        name: web-app
        image: my-web-app 
        state: started 
        published_ports:
          - "8080:80"

    - name: Run Database App Container 
      docker_container: 
        name: database-app
        image: my-database-app 
        state: started 
        published_ports: 
          - "3306:3306"

Kubernetes - Deploying StatefulSet 
apiVersion: app/v1 
kind: Statefulset 
metadata: 
  name: pathnex-db 
spec: 
  serviceName: "pathnex-db" 
  replicas: 3 
  selector: 
    matchLabels: 
      app: pathnex-db 
  template: 
    metadata: 
      labels: 
        app: pathnex-db 
    spec: 
      containers: 
        - name: db 
          image: mysql:5.7
          volumeMounts: 
            - mountPath: /var/lib/mysql 
              nameL mysql-storage 
  volumeClaimTemplates: 
    - metadata: 
        name: mysql-storage 
      spec: 
        accessModes: ["ReadWriteOnce"]
        resources:
          requests: 
            storage: 1Gi

Jenkinsfile - Build and Deploy to Kubernetes 
pipeline {
    agent any
    environment {
        IMAGE_NAME = 'pathnex-web-app'
    }
    stages {
        stage('Build Docker Image') {
            steps {
                script {
                    docker.build("${env.IMAGW_NAMW}")
                }
            }
        }
        stage('Push Docker Image') {
            steps {
                script {
                    docker.withRegistry('https://docker.io', 'docker-credentials') {
                        docker.image("${env.IMAGE_NAMW}").push('latest')
                    }
                }
            }
        }
        stage('Deploy to Kubernetes') {
            steps {
                script {
                    sh 'kubectl apply -f deployment.yaml'
                }
            }
        }
    }
}
